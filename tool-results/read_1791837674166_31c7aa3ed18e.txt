   385→class Component extends DCLogic {
   386→  state = { h: 5, p: 3 };
   387→  renderVals() {
   388→    const fa = (n) => String(n).replace(/\d/g, (d) => '۰۱۲۳۴۵۶۷۸۹'[d]);
   389→    const money = () => '[قیمت]';
   390→    const price = Number(this.props.price ?? 890000);
   391→    const charm = Number(this.props.charm ?? 50000);
   392→    const done = Number(this.props.charityDone ?? 0);
   393→    const target = Math.max(1, Number(this.props.charityTarget ?? 30));
   394→    const { h, p } = this.state;
   395→    const all = [];
   396→    let hi = 0, pi = 0;
   397→    while (hi < h || pi < p) {
   398→      if (hi < h) { all.push('h'); hi++; }
   399→      if (pi < p) { all.push('p'); pi++; }
   400→    }
   401→    const cap = 26;
   402→    const shown = all.slice(0, cap).map((k, i) => ({
   403→      isHeart: k === 'h',
   404→      isPaw: k === 'p',
   405→      tr: 'display:inline-flex;transform:rotate(' + (((i * 37) % 28) - 14) + 'deg)',
   406→    }));
   407→    const doneNow = done + h + p;
   408→    return {
   409→      charm: money(charm),
   410→      hearts: fa(h),
   411→      paws: fa(p),
   412→      heartPlus: () => this.setState({ h: h + 1 }),
   413→      heartMinus: () => this.setState({ h: Math.max(0, h - 1) }),
   414→      pawPlus: () => this.setState({ p: p + 1 }),
   415→      pawMinus: () => this.setState({ p: Math.max(0, p - 1) }),
   416→      stickers: shown,
   417→      hasMore: all.length > cap,
   418→      more: fa(all.length - cap),
   419→      giftNo: fa(Number(this.props.nextNo ?? 101)),
   420→      doneNow: fa(doneNow),
   421→      target: '[هدف]',
   422→      barPct: Math.min(100, Math.round((doneNow / target) * 100)),
   423→      total: money(price + (h + p) * charm),
   424→    };
   425→  }
   426→}
