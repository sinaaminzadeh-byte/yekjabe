   408→    return {
   409→      charm: money(charm),
   410→      hearts: fa(h),
   411→      paws: fa(p),
   412→      heartPlus: () => this.setState({ h: h + 1 }),
   413→      heartMinus: () => this.setState({ h: Math.max(0, h - 1) }),
   414→      pawPlus: () => this.setState({ p: p + 1 }),

   421→      target: '[هدف]',
   422→      barPct: Math.min(100, Math.round((doneNow / target) * 100)),
