    91→========== 04-پنجه و قلب.html ==========
    92→class Component extends DCLogic {
    93→  state = { h: 5, p: 3 };
    94→  renderVals() {
    95→    const fa = (n) => String(n).replace(/\d/g, (d) => '۰۱۲۳۴۵۶۷۸۹'[d]);
    96→    const money = () => '[قیمت]';
    97→    const price = Number(this.props.price ?? 890000);
    98→    const charm = Number(this.props.charm ?? 50000);
    99→    const done = Number(this.props.charityDone ?? 0);
   100→    const target = Math.max(1, Number(this.props.charityTarget ?? 30));
   101→    const { h, p } = this.state;
   102→    const all = [];
   103→    let hi = 0, pi = 0;
   104→    while (hi < h || pi < p) {
   105→      if (hi < h) { all.push('h'); hi++; }
   106→      if (pi < p) { all.push('p'); pi++; }
   107→    }
   108→    const cap = 26;
   109→    const shown = all.slice(0, cap).map((k, i) => ({
   110→      isHeart: k === 'h',
   111→      isPaw: k === 'p',
   112→      tr: 'display:inline-flex;transform:rotate(' + (((i * 37) % 28) - 14) + 'deg)',
   113→    }));
   114→    const doneNow = done + h + p;
   115→    return {
   116→      charm: money(charm),
   117→      hearts: fa(h),
   118→      paws: fa(p),
   119→      heartPlus: () => this.setState({ h: h + 1 }),
   120→      heartMinus: () => this.setState({ h: Math.max(0, h - 1) }),
   121→      pawPlus: () => this.setState({ p: p + 1 }),
   122→      pawMinus: () => this.setState({ p: Math.max(0, p - 1) }),
   123→      stickers: shown,
   124→      hasMore: all.length > cap,
   125→      more: fa(all.length - cap),
   126→      giftNo: fa(Number(this.props.nextNo ?? 101)),
   127→      doneNow: fa(doneNow),
   128→      target: '[هدف]',
   129→      barPct: Math.min(100, Math.round((doneNow / target) * 100)),
   130→      total: money(price + (h + p) * charm),
   131→    };
   132→  }
   133→}
